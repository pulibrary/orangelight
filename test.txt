def send_to_libanswers
    http = Net::HTTP.new uri.host, uri.port
    http.use_ssl = true
    http.set_debug_output(Rails.logger)
    Rails.logger.info("Feedback form submission body: #{body}")
    http.post uri, body, { Authorization: "Bearer #{token}" }
  end

    private

      attr_reader :patron_name, :patron_email, :context, :title, :quid

      def body
        @body ||= data.to_a.map { |entry| "#{entry[0]}=#{CGI.escape(entry[1].to_s)}" }.join('&')
      end
